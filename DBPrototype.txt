-- LinkNoDin prototype database
--
-- Initial relational schema for project.
-- It covers candidates, companies, administrators, jobs, applications,
-- reviews, notifications, resources, courses, forums, plans, guides,
-- moderation and basic activity logs.
--

BEGIN;

-- =========================================================
-- 1. User accounts and roles
-- =========================================================

CREATE TABLE roles (
    id      SMALLINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    code    VARCHAR(30) NOT NULL UNIQUE,
    name    VARCHAR(80) NOT NULL
);

INSERT INTO roles (code, name) VALUES
    ('candidate', 'Candidate'),
    ('company', 'Company'),
    ('admin', 'Administrator')
ON CONFLICT (code) DO NOTHING;

CREATE TABLE users (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    role_id         SMALLINT NOT NULL REFERENCES roles(id),
    full_name       VARCHAR(160) NOT NULL,
    email           VARCHAR(254) NOT NULL,
    password_hash   TEXT NOT NULL,
    status          VARCHAR(20) NOT NULL DEFAULT 'active',
    email_verified  BOOLEAN NOT NULL DEFAULT false,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE password_reset_requests (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id         INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    token_hash      TEXT NOT NULL,
    expires_at      TIMESTAMPTZ NOT NULL,
    used_at         TIMESTAMPTZ,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- File metadata only; the actual file should live in local or cloud storage.
CREATE TABLE file_assets (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    uploaded_by     INT REFERENCES users(id) ON DELETE SET NULL,
    storage_key     TEXT NOT NULL UNIQUE,
    original_name   VARCHAR(255) NOT NULL,
    mime_type       VARCHAR(120),
    size_bytes      INT,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- =========================================================
-- 2. Candidate profiles and skills
-- =========================================================

CREATE TABLE candidate_profiles (
    user_id                 INT PRIMARY KEY REFERENCES users(id) ON DELETE CASCADE,
    phone                   VARCHAR(30),
    location                VARCHAR(160),
    occupation              VARCHAR(160),
    education_summary       TEXT,
    experience_years        NUMERIC(4,1),
    professional_summary    TEXT,
    cv_file_id              INT REFERENCES file_assets(id) ON DELETE SET NULL,
    updated_at              TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE skills (
    id      INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name    VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE candidate_skills (
    candidate_user_id   INT NOT NULL
                        REFERENCES candidate_profiles(user_id) ON DELETE CASCADE,
    skill_id            INT NOT NULL REFERENCES skills(id) ON DELETE CASCADE,
    proficiency         VARCHAR(40),
    PRIMARY KEY (candidate_user_id, skill_id)
);

-- =========================================================
-- 3. Companies
-- =========================================================

CREATE TABLE companies (
    id                  INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id             INT NOT NULL UNIQUE
                        REFERENCES users(id) ON DELETE CASCADE,
    commercial_name     VARCHAR(120) NOT NULL,
    legal_name          VARCHAR(160),
    sector              VARCHAR(100),
    employee_range      VARCHAR(50),
    founded_year        SMALLINT,
    department          VARCHAR(100),
    city                VARCHAR(100),
    address             VARCHAR(180),
    public_email        VARCHAR(254),
    public_phone        VARCHAR(30),
    website             VARCHAR(240),
    linkedin_url        VARCHAR(240),
    description         TEXT,
    culture             TEXT,
    logo_file_id        INT REFERENCES file_assets(id) ON DELETE SET NULL,
    verification_status VARCHAR(30) NOT NULL DEFAULT 'pending',
    created_at          TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- =========================================================
-- 4. Jobs and application questions
-- =========================================================

CREATE TABLE jobs (
    id                          INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    company_id                  INT NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
    created_by                  INT REFERENCES users(id) ON DELETE SET NULL,

    title                       VARCHAR(120) NOT NULL,
    openings                    INTEGER NOT NULL DEFAULT 1,
    contract_type               VARCHAR(60),
    modality                    VARCHAR(30),
    experience_level            VARCHAR(60),
    education_required          VARCHAR(100),
    work_time                   VARCHAR(60),
    schedule                    VARCHAR(60),
    work_days_hours             VARCHAR(150),
    area                        VARCHAR(100),

    summary                     TEXT,
    responsibilities            TEXT,
    license_requirement         VARCHAR(60),
    vehicle_requirement         VARCHAR(60),
    competencies                TEXT,
    requirements                TEXT,
    benefits                    TEXT,

    salary_min                  NUMERIC(12,2),
    salary_max                  NUMERIC(12,2),
    salary_period               VARCHAR(40),
    salary_confidential         BOOLEAN NOT NULL DEFAULT false,
    application_deadline        DATE,

    department                  VARCHAR(100),
    city                        VARCHAR(100),
    address                     VARCHAR(180),
    reference_point             VARCHAR(180),
    latitude                    NUMERIC(10,7),
    longitude                   NUMERIC(10,7),
    approximate_location_only   BOOLEAN NOT NULL DEFAULT true,

    contact_email               VARCHAR(254),
    application_method          VARCHAR(30) DEFAULT 'portal',
    application_instructions    TEXT,

    status                      VARCHAR(30) NOT NULL DEFAULT 'active',
    published_at                TIMESTAMPTZ,
    created_at                  TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at                  TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE job_questions (
    id          INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    job_id      INT NOT NULL REFERENCES jobs(id) ON DELETE CASCADE,
    question    TEXT NOT NULL,
    sort_order  SMALLINT NOT NULL DEFAULT 1,
    is_required BOOLEAN NOT NULL DEFAULT true
);

-- =========================================================
-- 5. Applications, answers and application status history
-- =========================================================

CREATE TABLE applications (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    job_id          INT NOT NULL REFERENCES jobs(id) ON DELETE CASCADE,
    candidate_id    INT NOT NULL
                    REFERENCES candidate_profiles(user_id) ON DELETE CASCADE,
    status          VARCHAR(30) NOT NULL DEFAULT 'received',
    cover_letter    TEXT,
    applied_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now(),

    -- Prevent duplicate applications to the same job by the same candidate.
    UNIQUE (job_id, candidate_id)
);

CREATE TABLE application_answers (
    id                  INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    application_id      INT NOT NULL REFERENCES applications(id) ON DELETE CASCADE,
    job_question_id     INT NOT NULL REFERENCES job_questions(id) ON DELETE CASCADE,
    answer              TEXT NOT NULL,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE application_status_history (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    application_id  INT NOT NULL REFERENCES applications(id) ON DELETE CASCADE,
    changed_by      INT REFERENCES users(id) ON DELETE SET NULL,
    status          VARCHAR(30) NOT NULL,
    note            TEXT,
    changed_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- =========================================================
-- 6. Saved jobs and profile views
-- =========================================================

CREATE TABLE saved_jobs (
    candidate_id    INT NOT NULL
                    REFERENCES candidate_profiles(user_id) ON DELETE CASCADE,
    job_id          INT NOT NULL REFERENCES jobs(id) ON DELETE CASCADE,
    saved_at        TIMESTAMPTZ NOT NULL DEFAULT now(),
    PRIMARY KEY (candidate_id, job_id)
);

CREATE TABLE candidate_profile_views (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    candidate_id    INT NOT NULL
                    REFERENCES candidate_profiles(user_id) ON DELETE CASCADE,
    viewer_user_id  INT REFERENCES users(id) ON DELETE SET NULL,
    viewed_at       TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE company_profile_views (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    company_id      INT NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
    viewer_user_id  INT REFERENCES users(id) ON DELETE SET NULL,
    viewed_at       TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- =========================================================
-- 7. Company reviews and home-page testimonials
-- =========================================================

-- Reviews are tied to a registered account. display_name is a snapshot
-- for showing the name used when the review was submitted.
CREATE TABLE company_reviews (
    id                  INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    company_id          INT NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
    user_id             INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    display_name        VARCHAR(100) NOT NULL,
    rating              SMALLINT NOT NULL,
    comment             TEXT NOT NULL,
    status              VARCHAR(30) NOT NULL DEFAULT 'visible',
    created_at          TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- User-submitted or administrator-managed testimonials are separate
-- from employer reviews because the home-page carousel is a different feature.
CREATE TABLE testimonials (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id         INT REFERENCES users(id) ON DELETE SET NULL,
    display_name    VARCHAR(100) NOT NULL,
    quote           TEXT NOT NULL,
    attribution     VARCHAR(160),
    is_verified     BOOLEAN NOT NULL DEFAULT false,
    status          VARCHAR(30) NOT NULL DEFAULT 'pending',
    sort_order      SMALLINT NOT NULL DEFAULT 1,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- =========================================================
-- 8. Notifications and preferences
-- =========================================================

CREATE TABLE notifications (
    id                  INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id             INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    notification_type   VARCHAR(50),
    title               VARCHAR(180) NOT NULL,
    message             TEXT NOT NULL,
    target_url          TEXT,
    is_read             BOOLEAN NOT NULL DEFAULT false,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE notification_preferences (
    id                  INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id             INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    notification_type   VARCHAR(50) NOT NULL,
    in_app_enabled      BOOLEAN NOT NULL DEFAULT true,
    email_enabled       BOOLEAN NOT NULL DEFAULT false,
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT now(),
    UNIQUE (user_id, notification_type)
);

-- =========================================================
-- 9. Resources, bookmarks and reading activity
-- =========================================================

CREATE TABLE resources (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    slug            VARCHAR(140) NOT NULL UNIQUE,
    title           VARCHAR(220) NOT NULL,
    category        VARCHAR(100),
    summary         TEXT,
    intro           TEXT,
    reading_minutes SMALLINT,
    status          VARCHAR(30) NOT NULL DEFAULT 'published',
    created_by      INT REFERENCES users(id) ON DELETE SET NULL,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE resource_sections (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    resource_id     INT NOT NULL REFERENCES resources(id) ON DELETE CASCADE,
    heading         VARCHAR(220),
    body            TEXT NOT NULL,
    sort_order      SMALLINT NOT NULL DEFAULT 1
);

CREATE TABLE resource_checklist_items (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    resource_id     INT NOT NULL REFERENCES resources(id) ON DELETE CASCADE,
    item_text       TEXT NOT NULL,
    sort_order      SMALLINT NOT NULL DEFAULT 1
);

CREATE TABLE user_resource_activity (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id         INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    resource_id     INT NOT NULL REFERENCES resources(id) ON DELETE CASCADE,
    is_saved        BOOLEAN NOT NULL DEFAULT false,
    read_at         TIMESTAMPTZ,
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    UNIQUE (user_id, resource_id)
);

-- =========================================================
-- 10. Courses and progress
-- =========================================================

CREATE TABLE courses (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    slug            VARCHAR(140) NOT NULL UNIQUE,
    title           VARCHAR(220) NOT NULL,
    description     TEXT,
    level           VARCHAR(60),
    status          VARCHAR(30) NOT NULL DEFAULT 'published',
    premium_only    BOOLEAN NOT NULL DEFAULT true,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE course_skills (
    course_id       INT NOT NULL REFERENCES courses(id) ON DELETE CASCADE,
    skill_id        INT NOT NULL REFERENCES skills(id) ON DELETE CASCADE,
    PRIMARY KEY (course_id, skill_id)
);

CREATE TABLE course_modules (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    course_id       INT NOT NULL REFERENCES courses(id) ON DELETE CASCADE,
    title           VARCHAR(220) NOT NULL,
    description     TEXT,
    content         TEXT,
    sort_order      SMALLINT NOT NULL DEFAULT 1
);

CREATE TABLE course_enrollments (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id         INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    course_id       INT NOT NULL REFERENCES courses(id) ON DELETE CASCADE,
    enrolled_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
    completed_at    TIMESTAMPTZ,
    UNIQUE (user_id, course_id)
);

CREATE TABLE course_module_progress (
    id              INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id         INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    module_id       INT NOT NULL REFERENCES course_modules(id) ON DELETE CASCADE,
    completed_at    TIMESTAMPTZ,
    UNIQUE (user_id, module_id)
);

-- =========================================================
-- 11. Forum topics and replies
-- =========================================================

CREATE TABLE forum_categories (
    id          INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

CREATE TABLE forum_replies (

    author_id       INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,

    status          VARCHAR(30) NOT NULL DEFAULT 'visible',
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE plans (

-- =========================================================
-- 14. Moderation reports, rules and actions

CREATE TABLE content_reports (

CREATE TABLE moderation_actions (

);

-- =========================================================
-- 16. Dashboard views
-- =========================================================

-- These numbers are derived from application data rather than maintained
-- as manually updated counters.

CREATE VIEW candidate_dashboard_stats AS
SELECT
    cp.user_id AS candidate_id,
    (SELECT count(*) FROM applications a
     WHERE a.candidate_id = cp.user_id) AS applications_sent,
    (SELECT count(*) FROM applications a
     WHERE a.candidate_id = cp.user_id AND a.status = 'in_review') AS applications_in_review,
    (SELECT count(*) FROM saved_jobs sj
     WHERE sj.candidate_id = cp.user_id) AS saved_jobs,
    (SELECT count(*) FROM candidate_profile_views pv
     WHERE pv.candidate_id = cp.user_id) AS profile_views
FROM candidate_profiles cp;

CREATE VIEW company_dashboard_stats AS
SELECT
    c.id AS company_id,
    (SELECT count(*) FROM jobs j
     WHERE j.company_id = c.id AND j.status = 'active') AS active_jobs,
    (SELECT count(*) FROM applications a
     JOIN jobs j ON j.id = a.job_id
     WHERE j.company_id = c.id) AS applications_received,
    (SELECT count(*) FROM applications a
     JOIN jobs j ON j.id = a.job_id
     WHERE j.company_id = c.id AND a.status = 'in_review') AS applications_in_review,
    (SELECT count(*) FROM company_profile_views pv
     WHERE pv.company_id = c.id) AS profile_views
FROM companies c;

CREATE VIEW admin_dashboard_stats AS
SELECT
    (SELECT count(*) FROM users) AS registered_users,
    (SELECT count(*) FROM companies) AS companies,
    (SELECT count(*) FROM jobs) AS jobs,
    (SELECT count(*) FROM applications) AS applications,
    (SELECT count(*) FROM content_reports WHERE status = 'open') AS open_reports;

COMMIT;
